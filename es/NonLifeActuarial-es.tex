	El actuario de seguros no de vida evalúa principalmente las 
	siguientes provisiones:
	\begin{itemize}
		\item Provisiones para siniestros pendientes de pago (PSAP, \emph{Reservas para siniestros declarados pero no liquidados (RBNS)}) 
		\item Provisiones para siniestros aún no manifestados (PSNEM, \emph{Reserva para siniestros incurridos pero no declarados (IBNR)})
		\item Provisiones para primas no devengadas (PPNA, \emph{Reserves for unearned premiums})
		\item Provisiones para riesgos en curso (PREC,  \emph{Reserves for outstanding risks (non-life)}
		)
	\end{itemize}

\begin{f}[Escalera en cadena determinista]
	%
	Sea \(C_{i k}\) el importe, acumulado hasta el año de desarrollo \(k\), de los siniestros ocurridos en el año del accidente \(i\), para \(1 \leq i, k \leq n\). \(C_{i k}\) puede representar bien el importe pagado, bien el coste total estimado (pago ya efectuado más reserva) del siniestro. Los importes \(C_{i k}\) se conocen para \(i+k \leq n+1\) y se busca estimar los valores de \(C_{i k}\) para \(i+k>n+1\), y en particular los valores finales \(C_{i n}\) para \(2 \leq i \leq n\). Estas notaciones se ilustran en el siguiente triángulo:
	%	\[
	%	C=\left(\begin{array}{ccccc}
		%	\rowcolor{white}	C_{1,1} & C_{1,2} & \cdots & C_{1, n-1} & C_{1, n} \\
		%	\rowcolor{white}	C_{2,1} & C_{2,2} & \cdots & C_{2, n-1} & \\
		%	\rowcolor{white}	\vdots & \vdots & \ddots & & \\

		%	\rowcolor{white}	C_{n, 1} & & & 
		%	\end{array}\right)
	%	\]
	
	El método de la escalera de cadenas estima los valores desconocidos, \(C_{i k}\) para \(i+k>n+1\), mediante
	\begin{equation}\label{CL1}
		\hat{C}_{i k}=C_{i, n+1-i} \cdot \hat{f}_{n+1-i} \cdots \hat{f}_{k-1} \quad i+k>n+1
	\end{equation}	ó
	\begin{equation}\label{CL2}
		\hat{f}_{k}=\frac{\color{OrangeProfondIRA}\sum_{i=1}^{n-k} C_{i, k+1}}{\color{BleuProfondIRA}\sum_{i=1}^{n-k} C_{i k}} \quad 1 \leq k \leq n-1 .
	\end{equation}
	La provisión para siniestros correspondiente al año del accidente (\(R_{i}\), \(2 \leq i \leq n\)) se calcula entonces mediante
	\begin{align*}
		\hat{R}_{i}=&C_{in }-C_{i, n+1-i}\\
		&=C_{i, n+1-i} \cdot \hat{f}_{n+1-i} \cdots \hat{f}_{n-1}-C_{i, n+1-i} 
	\end{align*}

\begin{f}[Método de Mack]
	
Las dos primeras hipótesis son las siguientes:
%
\begin{equation}\label{CL3}
E\left(C_{i, k+1} \mid C_{i 1}, \ldots, C_{i k}\right)=C_{i k} f_{k} \quad 1 \leq i \leq n, 1 \leq k \leq n-1
\end{equation}	
\begin{equation}\label{CL4}
\left\{C_{i 1}, \ldots, C_{i n}\right\},\left\{C_{j 1}, \ldots, C_{j n}\right\} \quad \forall i, j \quad \text{sont indépendants}
\end{equation}


Mack demuestra que, si se estiman los parámetros del modelo (\ref{CL3}) mediante (\ref{CL2}), dicho modelo estocástico (\ref{CL3}), combinado con la hipótesis (\ref{CL4}), proporciona exactamente las mismas reservas que el método original de Chain Ladder (\ref{CL1}).

Con la notación \(f_{i, k}= \frac{C_{i, k+1}}{C_{i, k}}\),  \(\hat{f}_{k}\) es la media de los \({\color{OrangeProfondIRA}f_{i, k}}\) ponderada por los \({\color{BleuProfondIRA} C_{i, k}}\):
\begin{align*}
	\hat{f}_{k}&=\frac{\sum_{i=1}^{n-k}{\color{BleuProfondIRA} C_{i, k}}\times \color{OrangeProfondIRA}f_{i, k}}{\color{BleuProfondIRA}\sum_{i=1}^{n-k} C_{i k}}
\end{align*}	
La varianza se expresa como:
\begin{align*}
	\hat{\sigma}_{k}^{2}&=\frac{1}{n-k-1} \sum_{i=1}^{n-k} C_{i k}\left(\frac{C_{i, k+1}}{C_{i k}}-\hat{f}_{k}\right)^{2} \\
	&=\frac{1}{n-k-1} \sum_{i=1}^{n-k}  \left(\frac{\color{BleuProfondIRA} C_{i, k+1}-{\color{BleuProfondIRA} C_{i, k}} \hat{f}_{k}}{\color{BleuProfondIRA} \sqrt{C_{i, k}}}\right)^{2} 
\end{align*}

\begin{f}[El modelo de riesgo colectivo]
	¿El modelo colectivo es el modelo básico en actuaría de no vida?
	\(X_{i}\) representa el importe del \(i\)-ésimo siniestro, \(N\) representa el número de siniestros y \(S\) el importe total a lo largo de un año
	\[
	S=\sum_{i=1}^{N} X_{i}
	\]
	sabiendo que \(S=0\) cuando \(N=0\) y que \(\left\{X_{i}\right\}_{i=1}^{\infty}\) es una secuencia \(iid\) y \(N \perp \left\{X_{i}\right\}_{i=1}^{\infty}\). 
	La dificultad radica en obtener la distribución de \(S\), a pesar de que \(\E [N]\) no es grande en el sentido de la TCL.
\end{f}



\begin{f}[La distribución de \(S\)]
	
	Sea \(G(x)=\mathbb{P}(S \leq x)\), \(F(x)=\mathbb{P}\left(X_{1} \leq x\right)\) y \(p_{n}=\mathbb{P}(N=n)\), de modo que \(\left\{p_{n}\right\}_{n=0}^{\infty}\) sea la función de probabilidad para el número de siniestros.
	
	\[
	\{S \leq x\}=\bigcup_{n=0}^{\infty}\{S \leq x \text { et } N=n\}
	\]
	\[
	\mathbb{P}(S \leq x \mid N=n)=\mathbb{P}\left(\sum_{i=1}^{n} X_{i} \leq x\right)=F^{n *}(x)
	\]
	Así pues, para \(x \geq 0\)
	
	\begin{equation*}\label{GxCollectif}
		G(x)=\sum_{n=0}^{\infty} p_{n} F^{n *}(x)
	\end{equation*}
	donde \(F^{n *}\) denota la convolución \(n\)-ésima; lamentablemente, esta no existe en forma cerrada para muchas distribuciones.    
	
	Si \(E[X]=m\)
	\[
	E[S]=E\left[N m\right]=E[N] m
	\]
	Este resultado es muy interesante: indica que el importe total esperado de los siniestros es el producto del número esperado de siniestros por el importe esperado de cada uno. Asimismo, dado que \(\left\{X_{i}\right\}_{i=1}^{\infty}\) son variables aleatorias independientes,
	\[
	V[S \mid N=n]=V\left[\sum_{i=1}^{n} X_{i}\right]=\sum_{i=1}^{n} V\left[X_{i}\right]
	\]
	\[
	\begin{aligned}
		V[S] &=E[V(S \mid N)]+V[E(S \mid N)] \\
		&=E[N] V\left[X_{i}\right]+V[N] m^{2}
	\end{aligned}
	\]    
\end{f}

\begin{f}[La clase de distribuciones \((a, b, 0)\)]
	
	Se dice que una distribución de recuento es \((a, b, 0)\) si su función de probabilidad \(\left\{p_{n}\right\}_{n=0}^{\infty}\) puede calcularse de forma recursiva a partir de la fórmula
	\[
	p_{n}=\left(a+\frac{b}{n}\right) p_{n-1}
	\]
	para \(n=1,2,3, \ldots,\) donde \(a\) y \(b\) son constantes.
	
	Existen exactamente tres distribuciones no triviales en la clase \((a, b, 0)\), a saber, la de Poisson, la binomial y la binomial negativa. Estos son los valores de \(a\) y \(b\) para las principales distribuciones \((a, b, 0)\):
	
	\begin{center}
		\begin{tabular}{ccc} 
			& \(a\) & \(b\) \\
			\hline\(\mathcal{P}_{ois}(\lambda)\) & 0 & \(\lambda\) \\
			\(\mathcal{B}_{in}(n, q)\) & \(-q /(1-q)\) & \((n+1) q /(1-q)\) \\
			\(\mathcal{N}\mathcal{B}_{in}(k, q)\) & \(1-q\) & \((1-q)(k-1)\) \\
			\(\mathcal{G}_{eo}( q)\) & \(1-q\) & 0 \\
			\textit{    Distribución de Panjer} & \(\frac{\lambda}{\alpha+\lambda}\) &    \(\frac {(\alpha -1)\lambda }{\alpha +\lambda }\) \\
			\hline 
		\end{tabular}
		
	\end{center}
	
	{\small\color{OrangeProfondIRA} La ley geométrica es un caso particular de la ley binomial negativa en la que k = 1.}
\end{f}

\begin{f}[Algoritmo de agregación de Panjer]
	
	El\textbf{algoritmo de Panjer} tiene como objetivo estimar la distribución de una ley compuesta de coste-frecuencia en condiciones particulares.
	\begin{itemize}
		\item \((X_i)_{i=1}^{N}\) variables discretas \(iid\) definidas en \(\{0,h,2h,3h...\}\)
		\item la ley del número en la clase denominada \((a,b,0)\)
	\end{itemize}
	
	
	Dado que suponemos ahora que los importes individuales de las solicitudes se distribuyen entre los enteros no negativos, se deduce que \(S\) también se distribuye entre los enteros no negativos. 
	Como \(S=\sum_{i=1}^{N} X_{i}\), se deduce que \(S=0\) si \(N=0\) o si \(N=n\) y \(\sum_{i=1}^{n} X_{i}=0\). Como \(\sum_{i=1}^{n} X_{i}=0\) únicamente si cada \(X_{i}=0\), se deduce, por independencia, que

	\[
	\mathbb{P}\left(\sum_{i=1}^{n} X_{i}=0\right)=f_{0}^{n}
	\]
	
\begin{equation*}
		\begin{cases}\label{Panjer0}\displaystyle
		g_{0}=p_{0}+\sum_{n=1}^{\infty} p_{n} f_{0}^{n}=P_{N}\left(f_{0}\right){\color{OrangeProfondIRA}\text{ si }a \ne 0},\\
	g_0=p_0\cdot \exp(f_0 b){\color{OrangeProfondIRA}\text{ si }a = 0,}\\
		g_{k}=\frac{1}{1-a f_{0}} \sum_{j=1}^{k}\left(a+\frac{b j}{k}\right) f_{j} g_{k-j}
		\end{cases}	
	\end{equation*}
\(g_{x}\) se expresa en función de \(g_{0}, g_{1}, \ldots, g_{x-1}\), por lo que su cálculo es recursivo y requiere un ordenador en la práctica. Sin embargo, la fórmula de Panjer evita calcular las convoluciones directas de \(g_{x}\), siendo mucho más eficiente computacionalmente. El algoritmo exige, eso sí, discretizar la variable \(X_i\).
\end{f}

\begin{f}[Panjer y la ley de Poisson]
	Cuando la frecuencia sigue una ley de Poisson, esto implica que {\color{OrangeProfondIRA} \(a=0\) y \(b=\lambda\).}
	El resultado se simplifica:
	\[
\begin{cases}
		g_0= e^{-\lambda (1-f_0)}
	\\
	g_k=\frac{\lambda}{k}\sum_{j=1}^k j . f_j  . g_{k-j}
\end{cases}	
\]
	
\end{f}

\begin{f}[Panjer y Pollaczeck-Khinchine-Beekman]
	
	Sea \(\tau_1\) el primer instante en el que \(R_t<\kappa(=\kappa_0)\). Entonces se establece que \(L_1=\kappa -R_{\tau_1}\).
	Se reinicia el proceso con \(\kappa_1=\kappa_0-R_{\tau_1}\) para hallar \(\tau_2\) y \(L_2=\kappa_1 -R_{\tau_2}\).
	Si continuamos de este modo, observamos que:
	\[
	M=\sup_{t\geq 0}\{ S_t -ct \} =\sum_{k=1}^{K}L_k
	\]
	donde \(K\sim \mathcal{G}eo(q)\) con \(q=1-\psi(0)\).  
	Al observar que las variables \(\left(  L_k\right)_{1\leq k\leq K}\) son \(iid\) (\(F\)), se tiene entonces que \(\psi(k)=\mathbb{P}[M>\kappa]\), dada por la fórmula de Pollaczeck-Khinchine-Beekman.
	
	
	La representación 
	\[
	\psi(\kappa)=\mathbb{P}\left[\sum_{j = 1}^{K}L_j>\kappa \right] 
	\]
	permite calcular la probabilidad de quiebra en un horizonte infinito mediante el algoritmo de Panjer.
		
\end{f}
